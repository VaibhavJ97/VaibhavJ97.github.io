"""Find sustainable BHE extraction rate using Brent's method.
Run for every 5 km pixel in Germany, across 8 CMIP6 models."""
from scipy.optimize import brentq
import numpy as np

def find_sustainable_q(t_years, params, t_min=-1.5):
    """Maximum q that keeps mean fluid temp above t_min
    over t_years, per SIA 384/6."""
    def temp_deficit(q):
        # Moving Finite Line Source (Rivera 2017)
        # + climate-driven surface trend (CMIP6)
        return fls_fluid_temp(q, t_years, params) - t_min

    # Brent's method: guaranteed convergence on sign change
    return brentq(temp_deficit, a=1.0, b=100.0, xtol=0.01)

geochat · api/chat.js Node.js · Vercel · Gemini 
// GeoChat serverless function: inject thesis context, call Gemini.
// Runs on Vercel free tier, costs €0/month at portfolio scale.
export default async function handler(req, res) {
  const { message, history = [] } = req.body;

  const r = await fetch(
    `${GEMINI_URL}?key=${process.env.GEMINI_API_KEY}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        // Thesis context injected server-side, user never sees it
        system_instruction: { parts: [{ text: THESIS_CONTEXT }] },
        // Last 6 turns of history + new user message
        contents: [
          ...history.slice(-6).map(t => ({
            role: t.role === 'user' ? 'user' : 'model',
            parts: [{ text: t.content }]
          })),
          { role: 'user', parts: [{ text: message }] }
        ],
        generationConfig: { temperature: 0.4, maxOutputTokens: 1500 }
      })
    }
  );

  const data = await r.json();
  const reply = data?.candidates?.[0]?.content?.parts?.[0]?.text;
  return res.status(200).json({ reply });
}
What this does: takes a chat message, injects the entire thesis as a system instruction, sends recent history for follow-up context, and asks Gemini for a grounded answer. The whole thing runs as a single Vercel serverless function.
