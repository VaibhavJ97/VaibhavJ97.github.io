// Serverless function: inject context, call the LLM, return a grounded reply.
// Runs on Vercel's free tier, costs €0/month at portfolio scale.
export default async function handler(req, res) {
  const { message, history = [] } = req.body;

  const r = await fetch(
    `${GEMINI_URL}?key=${process.env.GEMINI_API_KEY}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        // Context injected server-side, the user never sees it
        system_instruction: { parts: [{ text: KNOWLEDGE_BASE }] },
        // Last 6 turns of history + the new user message
        contents: [
          ...history.slice(-6).map(t => ({
            role: t.role === 'user' ? 'user' : 'model',
            parts: [{ text: t.content }]
          })),
          { role: 'user', parts: [{ text: message }] }
        ],
        generationConfig: { temperature: 0.4, maxOutputTokens: 1500 }
      })
    }
  );

  const data = await r.json();
  const reply = data?.candidates?.[0]?.content?.parts?.[0]?.text;
  return res.status(200).json({ reply });
}

"""Numeric solver run across a large grid of input cells.
Find the root of a target function for every pixel, robustly."""
from scipy.optimize import brentq
import numpy as np

def solve_cell(years, params, threshold=-1.5):
    """Largest input value that keeps the modeled output
    above a threshold over the given horizon."""
    def deficit(q):
        # model output minus the threshold we must stay above
        return model_output(q, years, params) - threshold

    # Brent's method: guaranteed convergence on a sign change
    return brentq(deficit, a=1.0, b=100.0, xtol=0.01)
