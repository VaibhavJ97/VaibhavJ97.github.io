// GeoChat serverless function: inject thesis context, call Gemini.
// Runs on Vercel free tier, costs €0/month at portfolio scale.
export default async function handler(req, res) {
  const { message, history = [] } = req.body;

  const r = await fetch(
    `${GEMINI_URL}?key=${process.env.GEMINI_API_KEY}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        // Thesis context injected server-side, user never sees it
        system_instruction: { parts: [{ text: THESIS_CONTEXT }] },
        // Last 6 turns of history + new user message
        contents: [
          ...history.slice(-6).map(t => ({
            role: t.role === 'user' ? 'user' : 'model',
            parts: [{ text: t.content }]
          })),
          { role: 'user', parts: [{ text: message }] }
        ],
        generationConfig: { temperature: 0.4, maxOutputTokens: 1500 }
      })
    }
  );

  const data = await r.json();
  const reply = data?.candidates?.[0]?.content?.parts?.[0]?.text;
  return res.status(200).json({ reply });
}
